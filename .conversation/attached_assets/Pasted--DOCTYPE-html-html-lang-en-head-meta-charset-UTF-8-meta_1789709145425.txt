  <script>
    const imageInput = document.getElementById('imageInput');
    const canvas = document.getElementById('photoCanvas');
    const ctx = canvas.getContext('2d');
    const canvasContainer = document.getElementById('canvasContainer');
    const tapInstruction = document.getElementById('tapInstruction');
    
    let lightHex = null;
    let darkHex = null;
    let pickState = 'light'; // 'light', 'dark', 'done'

    imageInput.addEventListener('change', (e) => {
      const file = e.target.files[0];
      if (!file) return;
      const img = new Image();
      img.onload = () => {
        canvas.width = img.width;
        canvas.height = img.height;
        ctx.drawImage(img, 0, 0);
        canvasContainer.style.display = 'block';
        pickState = 'light';
        tapInstruction.innerText = "Tap a LIGHT part of your face (e.g. Forehead/Cheek)";
      };
      img.src = URL.createObjectURL(file);
    });

    canvas.addEventListener('click', (e) => {
      const rect = canvas.getBoundingClientRect();
      const scaleX = canvas.width / rect.width;
      const scaleY = canvas.height / rect.height;
      const x = (e.clientX - rect.left) * scaleX;
      const y = (e.clientY - rect.top) * scaleY;

      const pixel = ctx.getImageData(x, y, 1, 1).data;
      const hex = "#" + ((1 << 24) + (pixel[0] << 16) + (pixel[1] << 8) + pixel[2]).toString(16).slice(1);

      if (pickState === 'light') {
        lightHex = hex;
        document.getElementById('lightSwatch').style.backgroundColor = hex;
        document.getElementById('lightHex').innerText = hex;
        pickState = 'dark';
        tapInstruction.innerText = "Now tap a DARK part of your face (e.g. Under jaw/contour zone)";
      } else if (pickState === 'dark') {
        darkHex = hex;
        document.getElementById('darkSwatch').style.backgroundColor = hex;
        document.getElementById('darkHex').innerText = hex;
        pickState = 'done';
        tapInstruction.innerText = "Colors recorded! Tap analyze below.";
        document.getElementById('analyzeBtn').style.display = 'block';
      }
    });

    function hexToRgb(hex) {
      const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
      return result ? { r: parseInt(result[1], 16), g: parseInt(result[2], 16), b: parseInt(result[3], 16) } : null;
    }

    function runFullAnalysis() {
      if (!lightHex || !darkHex) return;
      
      const rgbL = hexToRgb(lightHex);
      const rgbD = hexToRgb(darkHex);

      // Contrast Level
      const lightLum = 0.2126 * rgbL.r + 0.7152 * rgbL.g + 0.0722 * rgbL.b;
      const darkLum = 0.2126 * rgbD.r + 0.7152 * rgbD.g + 0.0722 * rgbD.b;
      const contrastDiff = lightLum - darkLum;
      const isHighContrast = contrastDiff > 45;

      // Undertone Detection
      const isWarm = (rgbL.r + rgbL.g) / 2 > rgbL.b + 10;
      let overtone = "Peach";
      if (rgbL.g > rgbL.r && rgbL.g > rgbL.b) overtone = "Olive";
      else if (!isWarm) overtone = "Cool Pink/Red";

      // Season Detection
      let season = "Bright Spring";
      if (isWarm && isHighContrast) season = "Warm Autumn";
      else if (!isWarm && isHighContrast) season = "Winter";
      else if (!isWarm && !isHighContrast) season = "Summer";

      const metal = isWarm ? "Gold Girlie ✨" : "Silver Girlie 💍";

      const html = `
        <div class="result-group"><div class="result-title">Undertone & Overtone</div>
        <div class="result-text">${isWarm ? 'Warm' : 'Cool'} Undertone with a ${overtone} hue.</div></div>

        <div class="result-group"><div class="result-title">Contrast Level</div>
        <div class="result-text">${isHighContrast ? 'High Contrast' : 'Low-to-Medium Contrast'} between skin, hair, and features.</div></div>

        <div class="result-group"><div class="result-title">Color Season</div>
        <div class="result-text"><strong>${season}</strong></div></div>

        <div class="result-group"><div class="result-title">Jewelry Match</div>
        <div class="result-text">You are a <strong>${metal}</strong>!</div></div>

        <div class="result-group"><div class="result-title">Colors to Avoid & Why</div>
        <div class="result-text">${isWarm ? 'Avoid muted muddy grays—they wash out your warm glow.' : 'Avoid bright mustard yellows—they make cool skin appear yellowed.'}</div></div>

        <div class="result-group"><div class="result-title">Makeup & Nails Recommendations</div>
        <div class="result-text">
          • <strong>Lip Balm:</strong> ${isWarm ? 'Tinted Coral or Warm Peach' : 'Berry Jam or Cool Mauve'}<br>
          • <strong>Blush:</strong> ${isWarm ? 'Warm Apricot' : 'Soft Rosy Pink'}<br>
          • <strong>Nail Polish:</strong> ${isWarm ? 'Creamy Nude or Terracotta' : 'Deep Wine or Pearl White'}<br>
          • <strong>Hair Dye:</strong> ${isWarm ? 'Warm Chestnut / Honey Blonde' : 'Ash Brown / Espresso'}
        </div></div>

        <div class="result-group"><div class="result-title">Hair Type & 3 Style Suggestions</div>
        <div class="result-text">
          • <strong>Layered Butterfly Cut:</strong> Adds volume around facial bone structure in a single fluid motion.<br>
          • <strong>Slicked Back High Pony:</strong> Emphasizes high facial contrast and sharp eye structure seamlessly.<br>
          • <strong>Soft Curtain Bangs:</strong> Frames soft facial bone contours and draws immediate focus to eyes.
        </div></div>
      `;

      document.getElementById('resultsContent').innerHTML = html;
      document.getElementById('resultsCard').style.display = 'block';
      document.getElementById('resultsCard').scrollIntoView({ behavior: 'smooth' });
    }

    function generateMakeupGuide() {
      const inspo = document.getElementById('makeupInspo').value || "Soft Glam";
      const products = document.getElementById('myMakeupProducts').value || "Concealer, Blush";

      document.getElementById('makeupGuide').innerHTML = `
        <strong>Guide for ${inspo}:</strong><br><br>
        <strong>Step 1:</strong> Prep skin and blend your base using ${products}.<br>
        <strong>Step 2:</strong> Apply blush along upper cheekbones to lift your bone structure.<br>
        <strong>Step 3:</strong> Tap lip balm onto the center of lips for a natural finish.<br><br>
        🎬 <strong>Video Tutorial:</strong> Search YouTube for <em>"${inspo} tutorial step by step"</em>.
      `;
    }

    function generateHairGuide() {
      const inspo = document.getElementById('hairInspo').value || "Volume Blowout";
      const tools = document.getElementById('myHairTools').value || "Brush, hair ties";

      document.getElementById('hairGuide').innerHTML = `
        <strong>Achieving ${inspo}:</strong><br><br>
        <strong>Step 1:</strong> Section hair using your existing setup (${tools}).<br>
        <strong>Step 2:</strong> Wrap individual sections outwards to sculpt framing layers.<br><br>
        🛒 <strong>Shopping Note:</strong> You already have necessary tools! No extra purchases required.
      `;
    }
